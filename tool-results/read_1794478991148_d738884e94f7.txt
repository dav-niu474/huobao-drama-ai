    92→// ── Batch Pipeline Manager (Singleton) ─────────────────────────

    94→class BatchPipelineManager {
    95→  private batches: Map<string, BatchState> = new Map()
    96→  private abortControllers: Map<string, AbortController> = new Map()
    97→  private pauseResolvers: Map<string, () => void> = new Map()
    98→  /**
    99→   * 发起批次的用户会话 Cookie（用于内部 API 调用鉴权）。
   100→   * 仅保存在服务端内存，不参与任何状态响应序列化。
   101→   */
   102→  private authCookies: Map<string, string> = new Map()
   103→
   104→  /**
   105→   * Start batch pipeline execution for a drama
   106→   */
   107→  async startBatch(
   108→    dramaId: string,
   109→    episodeIds?: string[],
   110→    steps?: string[],
   111→    options?: { authCookie?: string }
   112→  ): Promise<BatchState> {
   113→    // If there's already a running batch for this drama, return it
   114→    const existing = this.batches.get(dramaId)
   115→    if (existing && existing.status === 'running') {
   116→      return existing
   117→    }
   118→
   119→    // Get episodes
   120→    const whereClause: Record<string, unknown> = { dramaId }
   121→    if (episodeIds && episodeIds.length > 0) {
   122→      whereClause.id = { in: episodeIds }
   123→    }
   124→
   125→    const episodes = await db.episode.findMany({
   126→      where: whereClause,
   127→      orderBy: { episodeNumber: 'asc' },
   128→      select: {
   129→        id: true,
   130→        episodeNumber: true,
   131→        title: true,
   132→        rawContent: true,
   133→        scriptContent: true,
   134→        scriptStatus: true,
   135→        extractStatus: true,
   136→        storyboardStatus: true,
   137→        globalAssetsImported: true,
   138→        sourceChapterIds: true,
   139→      },
   140→    })
   141→
   142→    if (episodes.length === 0) {
   143→      throw new Error('No episodes found for this drama')
   144→    }
   145→
   146→    // Filter steps to execute
   147→    const stepsToExecute = steps && steps.length > 0
   148→      ? PIPELINE_STEPS_ORDER.filter((s) => steps!.includes(s))
   149→      : [...PIPELINE_STEPS_ORDER]
   150→
   151→    const totalSteps = stepsToExecute.length
   152→
   153→    // Create batch state
   154→    const batchState: BatchState = {
   155→      dramaId,
   156→      status: 'running',
   157→      totalEpisodes: episodes.length,
   158→      completedEpisodes: 0,
   159→      currentEpisodeIndex: 0,
   160→      currentStep: STEP_LABELS[stepsToExecute[0]] || '准备中',
   161→      episodes: episodes.map((ep) => ({
   162→        episodeId: ep.id,
   163→        episodeNumber: ep.episodeNumber,
   164→        title: ep.title,
   165→        status: 'pending',
   166→        completedSteps: 0,
   167→        totalSteps,
   168→        currentStep: null,
   169→      })),
   170→      startedAt: new Date(),
   171→    }
   172→
   173→    this.batches.set(dramaId, batchState)
   174→
   175→    // 保存发起者的会话 Cookie（内部 API 调用转发鉴权用）
   176→    if (options?.authCookie) {
   177→      this.authCookies.set(dramaId, options.authCookie)
   178→    }
   179→
   180→    // Create abort controller
   181→    const controller = new AbortController()
   182→    this.abortControllers.set(dramaId, controller)
   183→
   184→    // ── 解析剧集归属用户，并在其 ALS 上下文中执行整个批次 ──
   185→    // 后台执行会丢失请求级 AsyncLocalStorage 上下文，导致 aiClient
   186→    // 无法解析用户级供应商配置；这里显式恢复。
   187→    const drama = await db.drama.findUnique({
   188→      where: { id: dramaId },
   189→      select: { userId: true },
   190→    })
   191→    const dramaUserId = drama?.userId ?? undefined
   192→
   193→    const runExecution = () =>
   194→      this.executeBatch(dramaId, episodes, stepsToExecute, controller.signal)
   195→        .catch((err) => {
   196→          console.error(`[BatchPipeline] Batch execution failed:`, err)
   197→          const state = this.batches.get(dramaId)
   198→          if (state) {
   199→            state.status = 'failed'
   200→            state.error = err instanceof Error ? err.message : String(err)
   201→          }
   202→        })
   203→        .finally(() => {
   204→          // 批次结束后清理会话 Cookie，避免长期驻留内存
   205→          this.authCookies.delete(dramaId)
   206→        })
   207→
   208→    // Start execution in the background (non-blocking)
   209→    if (dramaUserId) {
   210→      userIdContext.run(dramaUserId, runExecution)
   211→    } else {
   212→      runExecution()
   213→    }
   214→
   215→    return batchState
   216→  }

   218→  /**
   219→   * Pause a running batch
   220→   */
   221→  async pauseBatch(dramaId: string): Promise<void> {
   222→    const state = this.batches.get(dramaId)
   223→    if (!state) throw new Error('No batch found for this drama')
   224→    if (state.status !== 'running') throw new Error('Batch is not running')
   225→
   226→    state.status = 'paused'
   227→    // The execution loop will check this and pause
   228→  }
   229→
   230→  /**
   231→   * Resume a paused batch
   232→   */
   233→  async resumeBatch(dramaId: string): Promise<void> {
   234→    const state = this.batches.get(dramaId)
   235→    if (!state) throw new Error('No batch found for this drama')
   236→    if (state.status !== 'paused') throw new Error('Batch is not paused')
   237→
   238→    state.status = 'running'
   239→
   240→    // Resolve the pause promise if one exists
   241→    const resolver = this.pauseResolvers.get(dramaId)
   242→    if (resolver) {
   243→      resolver()
   244→      this.pauseResolvers.delete(dramaId)
   245→    }
   246→  }

   248→  /**
   249→   * Get current batch status
   250→   */
   251→  getStatus(dramaId: string): BatchState | null {
   252→    return this.batches.get(dramaId) ?? null
   253→  }
   254→
   255→  /**
   256→   * Get batch status as API response format
   257→   */
   258→  getStatusResponse(dramaId: string): BatchStatusResponse | null {
   259→    const state = this.batches.get(dramaId)
   260→    if (!state) return null
   261→
   262→    const completedEpisodes = state.episodes.filter(
   263→      (e) => e.status === 'completed'
   264→    ).length
   265→
   266→    const totalStepsAll = state.episodes.reduce((s, e) => s + e.totalSteps, 0)
   267→    const completedStepsAll = state.episodes.reduce((s, e) => s + e.completedSteps, 0)
   268→    const progressPercent =
   269→      totalStepsAll > 0 ? Math.round((completedStepsAll / totalStepsAll) * 100) : 0
   270→
   271→    return {
   272→      batchId: state.dramaId,
   273→      status: state.status,
   274→      totalEpisodes: state.totalEpisodes,
   275→      completedEpisodes,
   276→      currentEpisode: state.currentEpisodeIndex + 1,
   277→      currentStep: state.currentStep,
   278→      progressPercent,
   279→      episodes: state.episodes,
   280→    }
   281→  }
   282→
   283→  // ── Internal: Authenticated internal API call ─────────────
   284→  // 之前直接 fetch 内部路由但不携带会话 Cookie，所有 requireAuth 的
   285→  // 生成路由都返回 401，制作步骤被静默跳过（当步骤已“完成”）。
   286→  // 现在转发发起批次的用户会话 Cookie，并在非 2xx 时抛出错误。
   287→  private async internalApi(
   288→    dramaId: string,
   289→    path: string,
   290→    body?: unknown
   291→  ): Promise<Record<string, unknown>> {
   292→    const headers: Record<string, string> = { 'Content-Type': 'application/json' }
   293→    const cookie = this.authCookies.get(dramaId)
   294→    if (cookie) headers['Cookie'] = cookie
   295→
   296→    const res = await fetch(`${getBaseUrl()}${path}`, {
   297→      method: 'POST',
   298→      headers,
   299→      body: body !== undefined ? JSON.stringify(body) : undefined,
   300→    })
   301→    if (!res.ok) {
   302→      const text = await res.text().catch(() => 'Unknown error')
   303→      throw new Error(`内部API调用失败 (HTTP ${res.status}) ${path}: ${text.slice(0, 200)}`)
   304→    }
   305→    return res.json().catch(() => ({}))
   306→  }

   308→  // ── Internal: Execute the batch ──────────────────────────────
   309→
   310→  private async executeBatch(
   311→    dramaId: string,
   312→    episodes: Array<{
   313→      id: string
   314→      episodeNumber: number
   315→      title: string
   316→      rawContent: string | null
   317→      scriptContent: string | null
   318→      scriptStatus: string
   319→      extractStatus: string
   320→      storyboardStatus: string
   321→      globalAssetsImported: boolean
   322→      sourceChapterIds: string | null
   323→    }>,
   324→    stepsToExecute: PipelineStep[],
   325→    signal: AbortSignal
   326→  ): Promise<void> {
   327→    const state = this.batches.get(dramaId)!
   328→    console.log(`[BatchPipeline] Starting batch for drama ${dramaId}: ${episodes.length} episodes, ${stepsToExecute.length} steps`)
   329→
   330→    for (let i = 0; i < episodes.length; i++) {
   331→      // Check abort
   332→      if (signal.aborted) {
   333→        state.status = 'failed'
   334→        state.error = 'Batch was aborted'
   335→        return
   336→      }
   337→
   338→      // Check pause
   339→      await this.checkPause(dramaId)
   340→      if (state.status === 'failed') return
   341→
   342→      const episode = episodes[i]
   343→      state.currentEpisodeIndex = i
   344→      state.episodes[i].status = 'running'
   345→
   346→      console.log(`[BatchPipeline] Processing episode ${episode.episodeNumber}: ${episode.title}`)
   347→
   348→      try {
   349→        await this.executeEpisode(episode, dramaId, stepsToExecute, signal, state, i)
   350→        state.episodes[i].status = 'completed'
   351→        state.completedEpisodes = i + 1
   352→      } catch (err) {
   353→        console.error(`[BatchPipeline] Episode ${episode.episodeNumber} failed:`, err)
   354→        state.episodes[i].status = 'failed'
   355→        state.episodes[i].error = err instanceof Error ? err.message : String(err)
   356→        // Continue with next episode
   357→        state.completedEpisodes = i + 1
   358→      }
   359→    }
   360→
   361→    // All episodes processed
   362→    state.status = state.episodes.every((e) => e.status === 'completed')
   363→      ? 'completed'
   364→      : state.episodes.some((e) => e.status === 'completed')
   365→        ? 'completed' // Mark as completed even if some failed
   366→        : 'failed'
   367→
   368→    state.currentStep = 'completed'
   369→    console.log(`[BatchPipeline] Batch completed for drama ${dramaId}: ${state.completedEpisodes}/${state.totalEpisodes} episodes`)
   370→  }

   372→  // ── Internal: Execute pipeline for a single episode ──────────
   373→
   374→  private async executeEpisode(
   375→    episode: {
   376→      id: string
   377→      episodeNumber: number
   378→      title: string
   379→      rawContent: string | null
   380→      scriptContent: string | null
   381→      scriptStatus: string
   382→      extractStatus: string
   383→      storyboardStatus: string
   384→      globalAssetsImported: boolean
   385→      sourceChapterIds: string | null
   386→    },
   387→    dramaId: string,
   388→    stepsToExecute: PipelineStep[],
   389→    signal: AbortSignal,
   390→    batchState: BatchState,
   391→    episodeIndex: number
   392→  ): Promise<void> {
   393→    for (let stepIdx = 0; stepIdx < stepsToExecute.length; stepIdx++) {
   394→      // Check abort
   395→      if (signal.aborted) throw new Error('Batch was aborted')
   396→
   397→      // Check pause
   398→      await this.checkPause(dramaId)
   399→      if (batchState.status === 'failed') throw new Error('Batch failed')
   400→
   401→      const step = stepsToExecute[stepIdx]
   402→      const stepLabel = STEP_LABELS[step]
   403→      batchState.currentStep = stepLabel
   404→      batchState.episodes[episodeIndex].currentStep = stepLabel
   405→
   406→      console.log(`[BatchPipeline] Episode ${episode.episodeNumber}, Step: ${stepLabel}`)
   407→
   408→      try {
   409→        await this.executeStep(episode, dramaId, step, signal)
   410→        batchState.episodes[episodeIndex].completedSteps = stepIdx + 1
   411→      } catch (err) {
   412→        const errMsg = err instanceof Error ? err.message : String(err)
   413→        console.error(`[BatchPipeline] Step ${stepLabel} failed for episode ${episode.episodeNumber}:`, err)
   414→        // Mark step as failed but continue to next step
   415→        // This is a soft failure — the episode continues
   416→        batchState.episodes[episodeIndex].completedSteps = stepIdx + 1
   417→        // 记录步骤错误明细，供前端展示与排查（不再静默丢弃）
   418→        const errors = batchState.episodes[episodeIndex].stepErrors || []
   419→        errors.push(`${stepLabel}: ${errMsg.slice(0, 300)}`)
   420→        batchState.episodes[episodeIndex].stepErrors = errors
   421→      }
   422→    }
   423→
   424→    batchState.episodes[episodeIndex].currentStep = null
   425→  }

   427→  // ── Internal: Execute a single pipeline step ─────────────────
   428→
   429→  private async executeStep(
   430→    episode: {
   431→      id: string
   432→      episodeNumber: number
   433→      title: string
   434→      rawContent: string | null
   435→      scriptContent: string | null
   436→      scriptStatus: string
   437→      extractStatus: string
   438→      storyboardStatus: string
   439→      globalAssetsImported: boolean
   440→      sourceChapterIds: string | null
   441→    },
   442→    dramaId: string,
   443→    step: PipelineStep,
   444→    _signal: AbortSignal
   445→  ): Promise<void> {
   446→    switch (step) {
   447→      case 'script:raw':
   448→        await this.executeRawContent(episode, dramaId)
   449→        break
   450→      case 'script:rewrite':
   451→        await this.executeScriptRewrite(episode, dramaId)
   452→        break
   453→      case 'script:extract':
   454→        await this.executeExtract(episode, dramaId)
   455→        break
   456→      case 'script:voice':
   457→        await this.executeVoiceAssign(episode, dramaId)
   458→        break
   459→      case 'script:storyboard':
   460→        await this.executeStoryboard(episode, dramaId)
   461→        break
   462→      case 'prod:chars':
   463→        await this.executeCharacterImages(episode, dramaId)
   464→        break
   465→      case 'prod:scenes':
   466→        await this.executeSceneImages(episode, dramaId)
   467→        break
   468→      case 'prod:dubbing':
   469→        await this.executeDubbing(episode, dramaId)
   470→        break
   471→      case 'prod:shots':
   472→        await this.executeShotFrames(episode, dramaId)
   473→        break
   474→      case 'prod:videos':
   475→        await this.executeVideos(episode, dramaId)
   476→        break
   477→      case 'prod:compose':
   478→        await this.executeCompose(episode, dramaId)
   479→        break
   480→      case 'export:merge':
   481→        await this.executeMerge(episode, dramaId)
   482→        break
   483→      default:
   484→        console.warn(`[BatchPipeline] Unknown step: ${step}`)
   485→    }
   486→  }

   488→  // ── Step implementations ─────────────────────────────────────
   489→
   490→  private async executeRawContent(
   491→    episode: { id: string; rawContent: string | null; globalAssetsImported: boolean; sourceChapterIds: string | null },
   492→    dramaId: string
   493→  ): Promise<void> {
   494→    // Check if rawContent is already filled
   495→    if (episode.rawContent?.trim()) return
   496→
   497→    // Step 1: Import global assets if not already done
   498→    if (!episode.globalAssetsImported) {
   499→      try {
   500→        // Import global assets — this also fills rawContent from novel if possible
   501→        const ep = await db.episode.findUnique({ where: { id: episode.id } })
   502→        if (ep && !ep.globalAssetsImported) {
   503→          await db.episode.update({
   504→            where: { id: episode.id },
   505→            data: { globalAssetsImported: true },
   506→          })
   507→        }
   508→      } catch (err) {
   509→        console.warn(`[BatchPipeline] Import global assets failed:`, err)
   510→      }
   511→    }
   512→
   513→    // Step 2: If still no rawContent, try to fill from novel chapters
   514→    const ep = await db.episode.findUnique({ where: { id: episode.id } })
   515→    if (ep?.rawContent?.trim()) return
   516→
   517→    if (ep?.sourceChapterIds) {
   518→      try {
   519→        let chapterIds: number[] = []
   520→        try {
   521→          chapterIds = JSON.parse(ep.sourceChapterIds)
   522→        } catch { chapterIds = [] }
   523→
   524→        if (chapterIds.length > 0) {
   525→          const novel = await db.novel.findUnique({ where: { dramaId } })
   526→          if (novel) {
   527→            let chapters: Array<{ index: number; title: string; content: string }> = []
   528→            try {
   529→              chapters = JSON.parse(novel.chapters)
   530→            } catch { chapters = [] }
   531→
   532→            const selected = chapters.filter((ch) => chapterIds.includes(ch.index))
   533→            if (selected.length > 0) {
   534→              const rawContent = selected
   535→                .map((ch) => `## ${ch.title}\n\n${ch.content}`)
   536→                .join('\n\n---\n\n')
   537→
   538→              await db.episode.update({
   539→                where: { id: episode.id },
   540→                data: { rawContent },
   541→              })
   542→            }
   543→          }
   544→        }
   545→      } catch (err) {
   546→        console.warn(`[BatchPipeline] Fill rawContent from novel failed:`, err)
   547→      }
   548→    }
   549→
   550→    // If still no rawContent, mark as skipped (we can't proceed without it)
   551→    const finalEp = await db.episode.findUnique({ where: { id: episode.id } })
   552→    if (!finalEp?.rawContent?.trim()) {
   553→      console.warn(`[BatchPipeline] Episode ${episode.id} has no rawContent — skipping script steps`)
   554→    }
   555→  }

   557→  private async executeScriptRewrite(
   558→    episode: { id: string; rawContent: string | null },
   559→    dramaId: string
   560→  ): Promise<void> {
   561→    // Check if already done
   562→    const ep = await db.episode.findUnique({ where: { id: episode.id } })
   563→    if (ep?.scriptContent?.trim()) return
   564→
   565→    // Need rawContent first
   566→    if (!ep?.rawContent?.trim()) {
   567→      console.warn(`[BatchPipeline] Skipping script rewrite — no rawContent`)
   568→      return
   569→    }
   570→
   571→    // ── 优先使用统一的 Toonflow 剧本生成（与单集工作台同一实现，保证格式一致）──
   572→    try {
   573→      const drama = await db.drama.findUnique({
   574→        where: { id: dramaId },
   575→        select: { userId: true },
   576→      })
   577→      await generateEpisodeScript(episode.id, {
   578→        duration: '120s',
   579→        userId: drama?.userId ?? undefined,
   580→      })
   581→      return
   582→    } catch (err) {
   583→      console.warn(
   584→        `[BatchPipeline] Toonflow script generation failed, falling back to script_rewriter agent:`,
   585→        err instanceof Error ? err.message : err
   586→      )
   587→    }
   588→

   796→    episode: { id: string },
   797→    dramaId: string
   798→  ): Promise<void> {
   799→    // Get storyboards without videos
   800→    const storyboards = await db.storyboard.findMany({
   801→      where: { episodeId: episode.id },
   802→    })
   803→    const withoutVideos = storyboards.filter((s) => !s.videoUrl)
   804→    if (withoutVideos.length === 0) return
   805→
   806→    // Generate videos for each storyboard
   807→    for (const sb of withoutVideos) {
   808→      try {
   809→        const data = await this.internalApi(dramaId, '/api/ai/generate-video', {
   810→          storyboardId: sb.id,
   811→          prompt: sb.videoPrompt || undefined,
   812→          firstFrameUrl: sb.firstFrameUrl || undefined,
   813→        })
   814→
   815→        // Async providers return { status: 'processing', taskId } — poll until
   816→        // completion, then persist the resulting videoUrl. Without this the
   817→        // batch pipeline marked async video tasks as done while nothing was
   818→        // ever written to the storyboard.
   819→        if (data.status === 'processing' && typeof data.taskId === 'string' && data.taskId) {
   820→          const videoUrl = await this.pollAsyncVideo(dramaId, data.taskId)
   821→          if (videoUrl) {
   822→            await db.storyboard.update({
   823→              where: { id: sb.id },
   824→              data: { videoUrl },
   825→            })
   826→          }
   827→        }
   828→      } catch (err) {
   829→        console.warn(`[BatchPipeline] Video failed for shot ${sb.shotNumber}:`, err)
   830→        throw err
   831→      }
   832→    }
   833→  }

   835→  /**
   836→   * Poll an async video generation task until completion.
   837→   * Returns the persisted video URL (file-storage path), or null on timeout.
   838→   */
   839→  private async pollAsyncVideo(
   840→    dramaId: string,
   841→    taskId: string,
   842→    intervalMs = 10000,
   843→    maxPolls = 60
   844→  ): Promise<string | null> {
   845→    for (let i = 0; i < maxPolls; i++) {
   846→      await new Promise((r) => setTimeout(r, intervalMs))
   847→      try {
   848→        const res = await this.internalApi(dramaId, '/api/ai/poll-status', {
   849→          category: 'video',
   850→          taskId,
   851→          dramaId,
   852→        })
   853→        if (res.status === 'completed') {
   854→          return typeof res.videoUrl === 'string' ? res.videoUrl : null
   855→        }
   856→        if (res.status === 'failed' || res.status === 'unsupported') {
   857→          throw new Error(String(res.error || '视频生成失败'))
   858→        }
   859→      } catch (err) {
   860→        if (i === maxPolls - 1) throw err
   861→      }
   862→    }
   863→    return null
   864→  }

   907→  // ── Internal: Check if batch is paused and wait ──────────────
   908→
   909→  private async checkPause(dramaId: string, signal?: AbortSignal): Promise<void> {
   910→    const state = this.batches.get(dramaId)
   911→    if (!state || state.status !== 'paused') return
   912→
   913→    // Wait until resumed
   914→    return new Promise<void>((resolve) => {
   915→      this.pauseResolvers.set(dramaId, resolve)
   916→      // Also set a timeout to re-check periodically
   917→      const interval = setInterval(() => {
   918→        const currentState = this.batches.get(dramaId)
   919→        if (!currentState || currentState.status !== 'paused') {
   920→          clearInterval(interval)
   921→          resolve()
   922→        }
   923→      }, 2000)
   924→
   925→      // Allow callers to abort the wait (e.g., when the batch is cancelled)
   926→      if (signal) {
   927→        signal.addEventListener('abort', () => {
   928→          clearInterval(interval)
   929→          this.pauseResolvers.delete(dramaId)
   930→          resolve()
   931→        }, { once: true })
   932→      }
   933→    })
   934→  }
   935→}
