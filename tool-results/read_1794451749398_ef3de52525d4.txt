    68→function relativeTime(dateStr: string, t: any): string {
    69→  const now = Date.now()
    70→  const then = new Date(dateStr).getTime()
    71→  const diff = Math.max(0, now - then)
    72→  const minutes = Math.floor(diff / 60000)
    73→  if (minutes < 1) return t('justNow')
    74→  if (minutes < 60) return t('minutesAgo', { count: minutes })
    75→  const hours = Math.floor(minutes / 60)
    76→  if (hours < 24) return t('hoursAgo', { count: hours })
    77→  const days = Math.floor(hours / 24)
    78→  if (days < 30) return t('daysAgo', { count: days })
    79→  const months = Math.floor(days / 30)
    80→  if (months < 12) return t('monthsAgo', { count: months })
    81→  return t('yearsAgo', { count: Math.floor(months / 12) })
    82→}

    84→function getStyleLabel(value: string, t: any): string {
    85→  const key = STYLE_ENTRIES.find(([v]) => v === value)?.[1]
    86→  return key ? t(key) : value
    87→}

   106→function ProjectCard({
   107→  drama,
   108→  onClick,
   109→  onDelete,
   110→  tc,
   111→  tp,
   112→}: {
   113→  drama: Drama
   114→  onClick: () => void

   117→  tp: any  // project namespace

   120→
   121→  const charCount = drama._count?.characters ?? 0
   122→  const sceneCount = drama._count?.scenes ?? 0
   123→  const epCount = drama._count?.episodes ?? 0
   124→  const totalEps = drama.totalEpisodes || 0
   125→  const progressPercent = totalEps > 0 ? Math.round((epCount / totalEps) * 100) : 0
   126→
   127→  return (
   128→    <motion.div
   129→      whileHover={{ y: -4 }}
   130→      onHoverStart={() => setHovered(true)}
   131→      onHoverEnd={() => setHovered(false)}
   132→    >
   133→      <Card
   134→        className="cursor-pointer group relative overflow-hidden border-border/60 hover:border-primary/60 hover:shadow-[0_0_16px_oklch(0.72_0.15_75/0.2)] transition-all duration-300 py-0 gap-0"
   135→        onClick={onClick}
   136→      >
   137→        <FilmStripSprockets />
   138→        <CardContent className="p-4 pt-3 flex flex-col gap-3">
   139→          {/* Title */}
   140→          <h3 className="text-base font-semibold leading-snug line-clamp-2 pr-6">
   141→            {drama.title}
   142→          </h3>
   143→
   144→          {/* Tags */}
   145→          <div className="flex flex-wrap gap-1.5">
   146→            <Badge variant="secondary" className="text-[11px] px-2 py-0">
   147→              {drama.genre}
   148→            </Badge>
   149→            <Badge variant="outline" className="text-[11px] px-2 py-0">
   150→              {getStyleLabel(drama.style, tp)}
   151→            </Badge>
   152→          </div>
   153→
   154→          {/* Meta */}
   155→          <div className="flex items-center gap-3 text-xs text-muted-foreground">
   156→            <span className="flex items-center gap-1">
   157→              <Users className="size-3" />{charCount}
   158→            </span>
   159→            <span className="flex items-center gap-1">
   160→              <MapPin className="size-3" />{sceneCount}
   161→            </span>
   162→            <span className="flex items-center gap-1">
   163→              <Film className="size-3" />{epCount}{tc('episodes')}
   164→            </span>
   165→          </div>
   166→
   167→          {/* Progress */}
   168→          {totalEps > 0 && (
   169→            <div className="space-y-1">
   170→              <Progress value={progressPercent} className="h-1.5" />
   171→              <p className="text-[10px] text-muted-foreground text-right">
   172→                {tp('percentComplete', { percent: progressPercent })}
   173→              </p>
   174→            </div>
   175→          )}
   176→
   177→          {/* Time */}
   178→          <div className="flex items-center gap-1 text-[11px] text-muted-foreground">
   179→            <Clock className="size-3" />
   180→            <span>{relativeTime(drama.updatedAt, tc)}</span>
   181→          </div>
   182→        </CardContent>
   183→
   184→        {/* Delete button */}
   185→        <Button
   186→          variant="ghost"
   187→          size="icon"
   188→          className={`absolute top-2 right-2 size-7 text-muted-foreground hover:text-destructive transition-opacity ${
   189→            hovered ? 'opacity-100' : 'opacity-0'
   190→          }`}
   191→          onClick={(e) => {
   192→            e.stopPropagation()
   193→            onDelete()
   194→          }}
   195→        >
   196→          <Trash2 className="size-3.5" />
   197→        </Button>
   198→      </Card>
   199→    </motion.div>
   200→  )
   201→}

   527→                  <span className={`text-xs ${idx === createStep ? 'text-foreground font-medium' : 'text-muted-foreground'}`}>
   528→                    {tp(key)}

   530→                </div>
   531→                {idx < 2 && <div className="w-8 h-px bg-border mx-1" />}

   584→                  <div className="grid grid-cols-2 gap-2">

   586→                      ['9:16', 'portrait', Smartphone],
   587→                      ['16:9', 'landscape', Monitor],
   588→                    ] as const).map(([value, key, Icon]) => (

   590→                        key={value}
   591→                        type="button"
   592→                        onClick={() => setAspectRatio(value)}

   597→                        }`}

   599→                        <Icon className="size-4" />
   600→                        {tp(key)}

   603→                  </div>
   604→                </div>
   605→
   606→                {/* Total Episodes + Episode Duration */}
   607→                <div className="grid grid-cols-2 gap-4">

   611→                      type="number"
   612→                      min={1}
   613→                      max={999}
   614→                      value={totalEpisodes}
   615→                      onChange={(e) => {
   616→                        const v = parseInt(e.target.value, 10)
   617→                        setTotalEpisodes(Number.isFinite(v) && v > 0 ? v : 1)

   619→                    />
   620→                  </div>
   621→                  <div className="space-y-2">
   622→                    <label className="text-sm font-medium">{tp('episodeDuration')}</label>

   626→                      </SelectTrigger>
   627→                      <SelectContent>
   628→                        {DURATION_OPTIONS.map((value, i) => (
   629→                          <SelectItem key={value} value={value}>
   630→                            {tp(DURATION_KEYS[i])}
   631→                          </SelectItem>
   632→                        ))}
   633→                      </SelectContent>
   634→                    </Select>
   635→                  </div>
   636→                </div>
   637→              </div>
   638→            )}
   639→
   640→            {createStep === 1 && (
   641→              <div className="space-y-4">
   642→                <div>
   643→                  <h4 className="text-sm font-medium">{tp('models')}</h4>

   645→                </div>

   647→                  ['llm', modelLlm, setModelLlm],
   648→                  ['image', modelImage, setModelImage],

   650→                  ['tts', modelTts, setModelTts],
   651→                ] as const).map(([category, value, setter]) => (

   657→                      category={category as AiCategory}

   660→                    />

   662→                ))}
   663→              </div>

   665→
   666→            {createStep === 2 && (

   668→                {/* Genre */}

   670→                  <label className="text-sm font-medium">{tp('genre')}</label>

   680→                      ))}

   687→                  <label className="text-sm font-medium">{tp('visualStyle')}</label>
   688→                  <Select value={newStyle} onValueChange={setNewStyle}>
   689→                    <SelectTrigger className="w-full">
   690→                      <SelectValue />
   691→                    </SelectTrigger>
   692→                    <SelectContent>
   693→                      {STYLE_ENTRIES.map(([value, key]) => (

   712→                          {tp(key)}
   713→                        </SelectItem>

   715→                    </SelectContent>
   716→                  </Select>

   718→              </div>

   720→          </div>
   721→
   722→          <DialogFooter className="gap-2">
   723→            <Button variant="outline" onClick={() => setCreateOpen(false)}>
   724→              {tc('cancel')}
   725→            </Button>
   726→            {createStep > 0 && (
   727→              <Button
   728→                variant="outline"
   729→                onClick={() => setCreateStep((s) => Math.max(0, s - 1))}
   730→                disabled={creating}
   731→              >
   732→                <ArrowLeft className="size-4" />
   733→                {tc('back')}
   734→              </Button>
   735→            )}
   736→            {createStep < 2 ? (
   737→              <Button
   738→                onClick={() => setCreateStep((s) => Math.min(2, s + 1))}
   739→                disabled={createStep === 0 ? !newProjectName.trim() : false}
   740→              >
   741→                {tc('next')}
   742→                <ArrowRight className="size-4" />
   743→              </Button>
   744→            ) : (
   745→              <Button onClick={handleCreate} disabled={!newProjectName.trim() || creating}>
   746→                {creating ? tc('creating') : tp('createProject')}
   747→              </Button>
   748→            )}
   749→          </DialogFooter>
   750→        </DialogContent>
   751→      </Dialog>

   753→      {/* ── Script Upload Dialog ───────────────────────────── */}
   754→      <ScriptUploadDialog

   757→        onSuccess={fetchDramas}

   769→          <AlertDialogFooter>
   770→            <AlertDialogCancel>{tc('cancel')}</AlertDialogCancel>
   771→            <AlertDialogAction
   772→              onClick={handleDelete}
   773→              disabled={deleting}
   774→              className="bg-destructive text-white hover:bg-destructive/90"
   775→            >
