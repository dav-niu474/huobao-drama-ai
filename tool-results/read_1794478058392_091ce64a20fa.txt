   134→   134→function formatToolArgs(args: Record<string, unknown>): string {
   135→   135→  try {
   136→   136→    // For array arguments, show count instead of full content
   137→   137→    const summarized: Record<string, unknown> = {}
   138→   138→    for (const [key, value] of Object.entries(args)) {
   139→   139→      if (Array.isArray(value)) {
   140→   140→        summarized[key] = `[${value.length}项]`
   141→   141→      } else if (typeof value === 'string' && value.length > 100) {
   142→   142→        summarized[key] = value.slice(0, 100) + '...'
   143→   143→      } else {
   144→   144→        summarized[key] = value
   145→   145→      }
   146→   146→    }
   147→   147→    return JSON.stringify(summarized, null, 2)
   148→   148→  } catch {
   149→   149→    return String(args)
   150→   150→  }
   151→   151→}
   152→   152→
   153→   153→function formatToolResult(result: unknown): string {
   154→   154→  if (!result || typeof result !== 'object') return String(result)
   155→   155→  try {
   156→   156→    const str = JSON.stringify(result, null, 2)
   157→   157→    // Truncate very long results
   158→   158→    if (str.length > 500) {
   159→   159→      return str.slice(0, 500) + '\n... (结果已截断)'
   160→   160→    }
   161→   161→    return str
   162→   162→  } catch {
   163→   163→    return String(result)
   164→   164→  }
   165→   165→}

   167→   167→// ── Time formatter ────────────────────────────────────────
   168→   168→
   169→   169→function formatTime(ms: number): string {
   170→   170→  if (ms < 1000) return `${ms}ms`
   171→   171→  if (ms < 60000) return `${(ms / 1000).toFixed(1)}s`
   172→   172→  return `${Math.floor(ms / 60000)}m ${Math.round((ms % 60000) / 1000)}s`
   173→   173→}
   174→   174→
   175→   175→function formatTimestamp(ts: number, startTs: number): string {
   176→   176→  const diff = ts - startTs
   177→   177→  if (diff < 1000) return `+${diff}ms`
   178→   178→  return `+${(diff / 1000).toFixed(1)}s`
   179→   179→}

   355→   355→export function AgentExecutionPanel({
   356→   356→  agentType,
   357→   357→  agentName,
   358→   358→  isRunning,
   359→   359→  logs,
   360→   360→  resultText,
   361→   361→  duration,
   362→   362→  error,
   363→   363→  defaultExpanded = true,
   364→   364→}: AgentExecutionPanelProps) {
   365→   365→  const [expanded, setExpanded] = useState(defaultExpanded)
   366→   366→  const scrollRef = useRef<HTMLDivElement>(null)
   367→   367→
   368→   368→  // Auto-scroll to bottom when new logs arrive
   369→   369→  useEffect(() => {
   370→   370→    if (scrollRef.current && expanded) {
   371→   371→      const el = scrollRef.current
   372→   372→      // Use requestAnimationFrame to ensure DOM has updated
   373→   373→      requestAnimationFrame(() => {
   374→   374→        el.scrollTop = el.scrollHeight
   375→   375→      })
   376→   376→    }
   377→   377→  }, [logs, expanded])
   378→   378→
   379→   379→  const startTimestamp = logs.length > 0 ? logs[0].timestamp : Date.now()
   380→   380→
   381→   381→  // Calculate stats
   382→   382→  const toolCallCount = logs.filter(l => l.type === 'tool_call').length
   383→   383→  const toolResultCount = logs.filter(l => l.type === 'tool_result').length
   384→   384→  const toolErrorCount = logs.filter(l => l.type === 'tool_error').length
   385→   385→  const totalSteps = logs.reduce((max, l) => Math.max(max, l.stepNumber || 0), 0)
   386→   386→
   387→   387→  // Get skill status from starting event
   388→   388→  const startEvent = logs.find(l => l.type === 'starting')
   389→   389→  const skillLoaded = startEvent?.skillLoaded
   390→   390→
   391→   391→  return (
   392→   392→    <Card className="border-primary/20 shadow-sm">
   393→   393→      <Collapsible open={expanded} onOpenChange={setExpanded}>
   394→   394→        <CollapsibleTrigger asChild>
   395→   395→          <CardHeader className="py-3 px-4 cursor-pointer hover:bg-muted/30 transition-colors">
   396→   396→            <div className="flex items-center justify-between">
   397→   397→              <div className="flex items-center gap-3">
   398→   398→                <div className={`rounded-full p-1.5 ${isRunning ? 'bg-primary/10' : error ? 'bg-red-100 dark:bg-red-900/30' : 'bg-emerald-100 dark:bg-emerald-900/30'}`}>
   399→   399→                  {isRunning ? (
   400→   400→                    <Loader2 className="size-4 text-primary animate-spin" />
   401→   401→                  ) : error ? (
   402→   402→                    <AlertCircle className="size-4 text-red-500" />
   403→   403→                  ) : logs.length > 0 && logs[logs.length - 1]?.type === 'completed' ? (
   404→   404→                    <Sparkles className="size-4 text-emerald-500" />
   405→   405→                  ) : (
   406→   406→                    <Brain className="size-4 text-muted-foreground" />
   407→   407→                  )}
   408→   408→                </div>
   409→   409→                <div>
   410→   410→                  <CardTitle className="text-sm font-semibold">{agentName}</CardTitle>
   411→   411→                  <div className="flex items-center gap-2 mt-0.5">
   412→   412→                    <span className="text-[10px] text-muted-foreground font-mono">{agentType}</span>
   413→   413→                    {skillLoaded !== undefined && (
   414→   414→                      <Badge variant="outline" className={`text-[9px] px-1 py-0 ${skillLoaded ? 'text-emerald-600 border-emerald-300' : 'text-amber-600 border-amber-300'}`}>
   415→   415→                        {skillLoaded ? '✓ SKILL已加载' : '⚠ SKILL未找到'}
   416→   416→                      </Badge>
   417→   417→                    )}
   418→   418→                  </div>
   419→   419→                </div>
   420→   420→              </div>
   421→   421→
   422→   422→              <div className="flex items-center gap-3">
   423→   423→                {/* Stats */}
   424→   424→                {logs.length > 0 && (
   425→   425→                  <div className="hidden sm:flex items-center gap-3 text-[10px] text-muted-foreground">
   426→   426→                    {totalSteps > 0 && (
   427→   427→                      <span className="flex items-center gap-1">
   428→   428→                        <RotateCcw className="size-3" />
   429→   429→                        {totalSteps}步
   430→   430→                      </span>
   431→   431→                    )}
   432→   432→                    {toolCallCount > 0 && (
   433→   433→                      <span className="flex items-center gap-1">
   434→   434→                        <Wrench className="size-3" />
   435→   435→                        {toolCallCount}次工具
   436→   436→                      </span>

   474→   474→                <AnimatePresence initial={false}>
   475→   475→                  {logs.map((entry, index) => (

   477→   477→                      key={entry.id}

   487→   487→                    </motion.div>

   489→   489→                </AnimatePresence>
   490→   490→
   491→   491→                {/* Empty state */}
   492→   492→                {logs.length === 0 && !isRunning && (
   493→   493→                  <div className="flex items-center justify-center py-8 text-muted-foreground">
   494→   494→                    <div className="text-center">

   542→   542→  )

   545→   545→// ── Hook: useAgentExecution ───────────────────────────────
   546→   546→// Manages the SSE stream connection and accumulates log entries
   547→   547→
   548→   548→export function useAgentExecution() {
   549→   549→  const [logs, setLogs] = useState<Record<string, AgentLogEntry[]>>({})
   550→   550→  const [runningAgents, setRunningAgents] = useState<Set<string>>(new Set())
   551→   551→  const [resultTexts, setResultTexts] = useState<Record<string, string>>({})
   552→   552→  const [durations, setDurations] = useState<Record<string, number>>({})
   553→   553→  const [errors, setErrors] = useState<Record<string, string | null>>({})
   554→   554→  const idCounter = useRef(0)
   555→   555→
   556→   556→  const startAgent = async (
   557→   557→    agentType: string,
   558→   558→    episodeId: string,
   559→   559→    dramaId: string,
   560→   560→    message: string,
   561→   561→    options?: { model?: string }
   562→   562→  ): Promise<{ hadErrors: boolean; toolErrors: string[] }> => {
   563→   563→    // Track tool errors during execution
   564→   564→    const toolErrors: string[] = []
   565→   565→    let hadTopLevelError = false
   566→   566→
   567→   567→    // Reset state for this agent
   568→   568→    setLogs(prev => ({ ...prev, [agentType]: [] }))
   569→   569→    setRunningAgents(prev => new Set(prev).add(agentType))
   570→   570→    setResultTexts(prev => ({ ...prev, [agentType]: '' }))
   571→   571→    setDurations(prev => ({ ...prev, [agentType]: 0 }))
   572→   572→    setErrors(prev => ({ ...prev, [agentType]: null }))
   573→   573→
   574→   574→    try {
   575→   575→      const res = await fetch(`/api/agent/${agentType}/stream`, {
   576→   576→        method: 'POST',
   577→   577→        headers: { 'Content-Type': 'application/json' },
   578→   578→        body: JSON.stringify({ episodeId, dramaId, message, model: options?.model }),
   579→   579→      })
   580→   580→
   581→   581→      if (!res.ok) {
   582→   582→        const text = await res.text().catch(() => 'Unknown error')
   583→   583→        setErrors(prev => ({ ...prev, [agentType]: `API ${res.status}: ${text}` }))
   584→   584→        setRunningAgents(prev => {
   585→   585→          const next = new Set(prev)
   586→   586→          next.delete(agentType)
   587→   587→          return next
   588→   588→        })
   589→   589→        return { hadErrors: true, toolErrors: [] }
   590→   590→      }
   591→   591→
   592→   592→      const reader = res.body?.getReader()
   593→   593→      if (!reader) {
   594→   594→        setErrors(prev => ({ ...prev, [agentType]: 'No readable stream' }))
   595→   595→        setRunningAgents(prev => {
   596→   596→          const next = new Set(prev)
   597→   597→          next.delete(agentType)
   598→   598→          return next
   599→   599→        })
   600→   600→        return { hadErrors: true, toolErrors: [] }
   601→   601→      }
   602→   602→
   603→   603→      const decoder = new TextDecoder()
   604→   604→      let buffer = ''
   605→   605→
   606→   606→      while (true) {
   607→   607→        const { done, value } = await reader.read()
   608→   608→        if (done) break
   609→   609→
   610→   610→        buffer += decoder.decode(value, { stream: true })
   611→   611→        const lines = buffer.split('\n')
   612→   612→        buffer = lines.pop() || ''
   613→   613→
   614→   614→        for (const line of lines) {
   615→   615→          if (line.startsWith('data: ')) {
   616→   616→            try {
   617→   617→              const event = JSON.parse(line.slice(6))
   618→   618→              const entryId = `log-${++idCounter.current}`
   619→   619→
   620→   620→              const entry: AgentLogEntry = {
   621→   621→                id: entryId,
   622→   622→                type: event.type || event.step || 'unknown',
   623→   623→                message: event.message || '',
   624→   624→                timestamp: event.timestamp || Date.now(),
   625→   625→                stepNumber: event.stepNumber,
   626→   626→                toolCall: event.toolCall,
   627→   627→                toolResult: event.toolResult,
   628→   628→                textOutput: event.textOutput,
   629→   629→                agentType: event.agentType,
   630→   630→                agentName: event.agentName,
   631→   631→                skillLoaded: event.skillLoaded,
   632→   632→              }
   633→   633→
   634→   634→              // ── Heartbeat / Streaming thinking handling ──
   635→   635→              // When we receive a "thinking" event that says "仍在生成中",
   636→   636→              // it's a heartbeat to keep the connection alive. We UPDATE
   637→   637→              // the last thinking entry's message instead of adding a new one.
   638→   638→              // For streaming thinking (actual content), we also update the
   639→   639→              // last thinking entry for the same step to show real-time progress.
   640→   640→              if (event.type === 'thinking') {
   641→   641→                const isHeartbeat = event.message?.includes('仍在生成中')
   642→   642→                const isInitialThink = event.message?.includes('正在思考')
   643→   643→                
   644→   644→                setLogs(prev => {
   645→   645→                  const currentLogs = prev[agentType] || []
   646→   646→                  // Find the last "thinking" or "text_stream" entry for this step
   647→   647→                  const lastThinkingIdx = [...currentLogs]
   648→   648→                    .reverse()
   649→   649→                    .findIndex(l => 
   650→   650→                      (l.type === 'thinking' || l.type === 'text_stream') && 
   651→   651→                      l.stepNumber === event.stepNumber
   652→   652→                    )
   653→   653→
   654→   654→                  if (lastThinkingIdx !== -1) {
   655→   655→                    const realIdx = currentLogs.length - 1 - lastThinkingIdx
   656→   656→                    const existing = currentLogs[realIdx]
   657→   657→                    // Update the existing thinking entry:
   658→   658→                    // - For heartbeat: just update the message
   659→   659→                    // - For streaming content: update message and mark as streaming
   660→   660→                    const updated = [...currentLogs]
   661→   661→                    updated[realIdx] = { 
   662→   662→                      ...existing, 
   663→   663→                      message: event.message,
   664→   664→                      // If it's actual content (not heartbeat/initial), mark as streaming
   665→   665→                      isStreamingThinking: !isHeartbeat && !isInitialThink,
   666→   666→                      type: isHeartbeat ? existing.type : 'thinking',
   667→   667→                    }
   668→   668→                    return { ...prev, [agentType]: updated }
   669→   669→                  }
   670→   670→                  // No previous thinking found, add as new entry
   671→   671→                  return { ...prev, [agentType]: [...currentLogs, entry] }
   672→   672→                })
   673→   673→                // Don't process further — it's a thinking update
   674→   674→                continue
   675→   675→              }
   676→   676→
   677→   677→              // Handle backward compatibility with old format (step field)
   678→   678→              if (!event.type && event.step) {
   679→   679→                entry.type = event.step
   680→   680→              }
   681→   681→
   682→   682→              setLogs(prev => ({
   683→   683→                ...prev,
   684→   684→                [agentType]: [...(prev[agentType] || []), entry],
   685→   685→              }))
   686→   686→
   687→   687→              // Handle completion
   688→   688→              if (entry.type === 'completed') {
   689→   689→                const resultText = event.textOutput || event.text || ''
   690→   690→                setResultTexts(prev => ({ ...prev, [agentType]: resultText }))
   691→   691→                if (event.duration) {
   692→   692→                  setDurations(prev => ({ ...prev, [agentType]: event.duration }))
   693→   693→                }
   694→   694→              }
   695→   695→
   696→   696→              // Handle error
   697→   697→              if (entry.type === 'error') {
   698→   698→                setErrors(prev => ({ ...prev, [agentType]: event.message }))
   699→   699→                hadTopLevelError = true
   700→   700→              }
   701→   701→
   702→   702→              // Track tool errors for caller to check
   703→   703→              if (entry.type === 'tool_error') {
   704→   704→                const errMsg = event.toolResult?.error || event.message || 'Unknown tool error'
   705→   705→                toolErrors.push(errMsg)
   706→   706→              }
   707→   707→            } catch {
   708→   708→              // ignore parse errors
   709→   709→            }
   710→   710→          }
   711→   711→        }
   712→   712→      }
   713→   713→    } catch (err) {
   714→   714→      hadTopLevelError = true
   715→   715→      setErrors(prev => ({
   716→   716→        ...prev,
   717→   717→        [agentType]: err instanceof Error ? err.message : String(err),
   718→   718→      }))
   719→   719→    } finally {
   720→   720→      setRunningAgents(prev => {
   721→   721→        const next = new Set(prev)
   722→   722→        next.delete(agentType)
   723→   723→        return next
   724→   724→      })
   725→   725→    }
   726→   726→
   727→   727→    return { hadErrors: hadTopLevelError || toolErrors.length > 0, toolErrors }
   728→   728→  }
   729→   729→
   730→   730→  const clearAgent = (agentType: string) => {
   731→   731→    setLogs(prev => {
   732→   732→      const next = { ...prev }
   733→   733→      delete next[agentType]
   734→   734→      return next
   735→   735→    })
   736→   736→    setResultTexts(prev => {
   737→   737→      const next = { ...prev }
   738→   738→      delete next[agentType]
   739→   739→      return next
   740→   740→    })
   741→   741→    setDurations(prev => {
   742→   742→      const next = { ...prev }
   743→   743→      delete next[agentType]
   744→   744→      return next
   745→   745→    })
   746→   746→    setErrors(prev => {
   747→   747→      const next = { ...prev }
   748→   748→      delete next[agentType]
   749→   749→      return next
   750→   750→    })
   751→   751→  }
   752→   752→
   753→   753→  const isRunning = (agentType: string) => runningAgents.has(agentType)
   754→   754→
   755→   755→  return {
   756→   756→    logs,
   757→   757→    runningAgents,
   758→   758→    resultTexts,
   759→   759→    durations,
   760→   760→    errors,
   761→   761→    startAgent,
   762→   762→    clearAgent,
   763→   763→    isRunning,
   764→   764→  }
   765→   765→}
