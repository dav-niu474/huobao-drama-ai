    94→    94→class BatchPipelineManager {
    95→    95→  private batches: Map<string, BatchState> = new Map()
    96→    96→  private abortControllers: Map<string, AbortController> = new Map()
    97→    97→  private pauseResolvers: Map<string, () => void> = new Map()
    98→    98→  /**
    99→    99→   * 发起批次的用户会话 Cookie（用于内部 API 调用鉴权）。
   100→   100→   * 仅保存在服务端内存，不参与任何状态响应序列化。
   101→   101→   */
   102→   102→  private authCookies: Map<string, string> = new Map()

   104→   104→  /**
   105→   105→   * Start batch pipeline execution for a drama
   106→   106→   */
   107→   107→  async startBatch(
   108→   108→    dramaId: string,
   109→   109→    episodeIds?: string[],
   110→   110→    steps?: string[],
   111→   111→    options?: { authCookie?: string }
   112→   112→  ): Promise<BatchState> {
   113→   113→    // If there's already a running batch for this drama, return it
   114→   114→    const existing = this.batches.get(dramaId)
   115→   115→    if (existing && existing.status === 'running') {
   116→   116→      return existing
   117→   117→    }
   118→   118→
   119→   119→    // Get episodes
   120→   120→    const whereClause: Record<string, unknown> = { dramaId }
   121→   121→    if (episodeIds && episodeIds.length > 0) {
   122→   122→      whereClause.id = { in: episodeIds }
   123→   123→    }
   124→   124→
   125→   125→    const episodes = await db.episode.findMany({
   126→   126→      where: whereClause,
   127→   127→      orderBy: { episodeNumber: 'asc' },
   128→   128→      select: {
   129→   129→        id: true,
   130→   130→        episodeNumber: true,
   131→   131→        title: true,
   132→   132→        rawContent: true,
   133→   133→        scriptContent: true,
   134→   134→        scriptStatus: true,
   135→   135→        extractStatus: true,
   136→   136→        storyboardStatus: true,
   137→   137→        globalAssetsImported: true,
   138→   138→        sourceChapterIds: true,
   139→   139→      },
   140→   140→    })
   141→   141→
   142→   142→    if (episodes.length === 0) {
   143→   143→      throw new Error('No episodes found for this drama')
   144→   144→    }
   145→   145→
   146→   146→    // Filter steps to execute
   147→   147→    const stepsToExecute = steps && steps.length > 0
   148→   148→      ? PIPELINE_STEPS_ORDER.filter((s) => steps!.includes(s))
   149→   149→      : [...PIPELINE_STEPS_ORDER]
   150→   150→
   151→   151→    const totalSteps = stepsToExecute.length
   152→   152→
   153→   153→    // Create batch state
   154→   154→    const batchState: BatchState = {
   155→   155→      dramaId,
   156→   156→      status: 'running',
   157→   157→      totalEpisodes: episodes.length,
   158→   158→      completedEpisodes: 0,
   159→   159→      currentEpisodeIndex: 0,
   160→   160→      currentStep: STEP_LABELS[stepsToExecute[0]] || '准备中',
   161→   161→      episodes: episodes.map((ep) => ({
   162→   162→        episodeId: ep.id,
   163→   163→        episodeNumber: ep.episodeNumber,
   164→   164→        title: ep.title,
   165→   165→        status: 'pending',
   166→   166→        completedSteps: 0,
   167→   167→        totalSteps,
   168→   168→        currentStep: null,
   169→   169→      })),
   170→   170→      startedAt: new Date(),
   171→   171→    }
   172→   172→
   173→   173→    this.batches.set(dramaId, batchState)
   174→   174→
   175→   175→    // 保存发起者的会话 Cookie（内部 API 调用转发鉴权用）
   176→   176→    if (options?.authCookie) {
   177→   177→      this.authCookies.set(dramaId, options.authCookie)
   178→   178→    }
   179→   179→
   180→   180→    // Create abort controller
   181→   181→    const controller = new AbortController()
   182→   182→    this.abortControllers.set(dramaId, controller)
   183→   183→
   184→   184→    // ── 解析剧集归属用户，并在其 ALS 上下文中执行整个批次 ──
   185→   185→    // 后台执行会丢失请求级 AsyncLocalStorage 上下文，导致 aiClient
   186→   186→    // 无法解析用户级供应商配置；这里显式恢复。
   187→   187→    const drama = await db.drama.findUnique({
   188→   188→      where: { id: dramaId },
   189→   189→      select: { userId: true },
   190→   190→    })
   191→   191→    const dramaUserId = drama?.userId ?? undefined
   192→   192→
   193→   193→    const runExecution = () =>
   194→   194→      this.executeBatch(dramaId, episodes, stepsToExecute, controller.signal)
   195→   195→        .catch((err) => {
   196→   196→          console.error(`[BatchPipeline] Batch execution failed:`, err)
   197→   197→          const state = this.batches.get(dramaId)
   198→   198→          if (state) {
   199→   199→            state.status = 'failed'
   200→   200→            state.error = err instanceof Error ? err.message : String(err)
   201→   201→          }
   202→   202→        })
   203→   203→        .finally(() => {
   204→   204→          // 批次结束后清理会话 Cookie，避免长期驻留内存
   205→   205→          this.authCookies.delete(dramaId)
   206→   206→        })
   207→   207→
   208→   208→    // Start execution in the background (non-blocking)
   209→   209→    if (dramaUserId) {
   210→   210→      userIdContext.run(dramaUserId, runExecution)
   211→   211→    } else {
   212→   212→      runExecution()
   213→   213→    }
   214→   214→
   215→   215→    return batchState
   216→   216→  }
   217→   217→
   218→   218→  /**
   219→   219→   * Pause a running batch
   220→   220→   */
   221→   221→  async pauseBatch(dramaId: string): Promise<void> {
   222→   222→    const state = this.batches.get(dramaId)
   223→   223→    if (!state) throw new Error('No batch found for this drama')
   224→   224→    if (state.status !== 'running') throw new Error('Batch is not running')
   225→   225→
   226→   226→    state.status = 'paused'
   227→   227→    // The execution loop will check this and pause
   228→   228→  }
   229→   229→
   230→   230→  /**
   231→   231→   * Resume a paused batch
   232→   232→   */
   233→   233→  async resumeBatch(dramaId: string): Promise<void> {
   234→   234→    const state = this.batches.get(dramaId)
   235→   235→    if (!state) throw new Error('No batch found for this drama')
   236→   236→    if (state.status !== 'paused') throw new Error('Batch is not paused')
   237→   237→
   238→   238→    state.status = 'running'
   239→   239→
   240→   240→    // Resolve the pause promise if one exists
   241→   241→    const resolver = this.pauseResolvers.get(dramaId)
   242→   242→    if (resolver) {
   243→   243→      resolver()
   244→   244→      this.pauseResolvers.delete(dramaId)
   245→   245→    }
   246→   246→  }

   255→   255→  /**
   256→   256→   * Get batch status as API response format
   257→   257→   */
   258→   258→  getStatusResponse(dramaId: string): BatchStatusResponse | null {
   259→   259→    const state = this.batches.get(dramaId)
   260→   260→    if (!state) return null
   261→   261→
   262→   262→    const completedEpisodes = state.episodes.filter(
   263→   263→      (e) => e.status === 'completed'
   264→   264→    ).length
   265→   265→
   266→   266→    const totalStepsAll = state.episodes.reduce((s, e) => s + e.totalSteps, 0)
   267→   267→    const completedStepsAll = state.episodes.reduce((s, e) => s + e.completedSteps, 0)
   268→   268→    const progressPercent =
   269→   269→      totalStepsAll > 0 ? Math.round((completedStepsAll / totalStepsAll) * 100) : 0
   270→   270→
   271→   271→    return {
   272→   272→      batchId: state.dramaId,
   273→   273→      status: state.status,
   274→   274→      totalEpisodes: state.totalEpisodes,
   275→   275→      completedEpisodes,
   276→   276→      currentEpisode: state.currentEpisodeIndex + 1,
   277→   277→      currentStep: state.currentStep,
   278→   278→      progressPercent,
   279→   279→      episodes: state.episodes,
   280→   280→    }
   281→   281→  }

   283→   283→  // ── Internal: Authenticated internal API call ─────────────
   284→   284→  // 之前直接 fetch 内部路由但不携带会话 Cookie，所有 requireAuth 的
   285→   285→  // 生成路由都返回 401，制作步骤被静默跳过（当步骤已“完成”）。
   286→   286→  // 现在转发发起批次的用户会话 Cookie，并在非 2xx 时抛出错误。
   287→   287→  private async internalApi(
   288→   288→    dramaId: string,
   289→   289→    path: string,
   290→   290→    body?: unknown
   291→   291→  ): Promise<Record<string, unknown>> {
   292→   292→    const headers: Record<string, string> = { 'Content-Type': 'application/json' }
   293→   293→    const cookie = this.authCookies.get(dramaId)
   294→   294→    if (cookie) headers['Cookie'] = cookie
   295→   295→
   296→   296→    const res = await fetch(`${getBaseUrl()}${path}`, {
   297→   297→      method: 'POST',
   298→   298→      headers,
   299→   299→      body: body !== undefined ? JSON.stringify(body) : undefined,
   300→   300→    })
   301→   301→    if (!res.ok) {
   302→   302→      const text = await res.text().catch(() => 'Unknown error')
   303→   303→      throw new Error(`内部API调用失败 (HTTP ${res.status}) ${path}: ${text.slice(0, 200)}`)
   304→   304→    }
   305→   305→    return res.json().catch(() => ({}))
   306→   306→  }

   308→   308→  // ── Internal: Execute the batch ──────────────────────────────
   309→   309→
   310→   310→  private async executeBatch(
   311→   311→    dramaId: string,
   312→   312→    episodes: Array<{
   313→   313→      id: string
   314→   314→      episodeNumber: number
   315→   315→      title: string
   316→   316→      rawContent: string | null
   317→   317→      scriptContent: string | null
   318→   318→      scriptStatus: string
   319→   319→      extractStatus: string
   320→   320→      storyboardStatus: string
   321→   321→      globalAssetsImported: boolean
   322→   322→      sourceChapterIds: string | null
   323→   323→    }>,
   324→   324→    stepsToExecute: PipelineStep[],
   325→   325→    signal: AbortSignal
   326→   326→  ): Promise<void> {
   327→   327→    const state = this.batches.get(dramaId)!
   328→   328→    console.log(`[BatchPipeline] Starting batch for drama ${dramaId}: ${episodes.length} episodes, ${stepsToExecute.length} steps`)
   329→   329→
   330→   330→    for (let i = 0; i < episodes.length; i++) {
   331→   331→      // Check abort
   332→   332→      if (signal.aborted) {
   333→   333→        state.status = 'failed'
   334→   334→        state.error = 'Batch was aborted'
   335→   335→        return
   336→   336→      }
   337→   337→
   338→   338→      // Check pause
   339→   339→      await this.checkPause(dramaId)
   340→   340→      if (state.status === 'failed') return
   341→   341→
   342→   342→      const episode = episodes[i]
   343→   343→      state.currentEpisodeIndex = i
   344→   344→      state.episodes[i].status = 'running'
   345→   345→
   346→   346→      console.log(`[BatchPipeline] Processing episode ${episode.episodeNumber}: ${episode.title}`)
   347→   347→
   348→   348→      try {
   349→   349→        await this.executeEpisode(episode, dramaId, stepsToExecute, signal, state, i)
   350→   350→        state.episodes[i].status = 'completed'
   351→   351→        state.completedEpisodes = i + 1
   352→   352→      } catch (err) {
   353→   353→        console.error(`[BatchPipeline] Episode ${episode.episodeNumber} failed:`, err)
   354→   354→        state.episodes[i].status = 'failed'
   355→   355→        state.episodes[i].error = err instanceof Error ? err.message : String(err)
   356→   356→        // Continue with next episode
   357→   357→        state.completedEpisodes = i + 1
   358→   358→      }
   359→   359→    }
   360→   360→
   361→   361→    // All episodes processed
   362→   362→    state.status = state.episodes.every((e) => e.status === 'completed')
   363→   363→      ? 'completed'
   364→   364→      : state.episodes.some((e) => e.status === 'completed')
   365→   365→        ? 'completed' // Mark as completed even if some failed
   366→   366→        : 'failed'
   367→   367→
   368→   368→    state.currentStep = 'completed'
   369→   369→    console.log(`[BatchPipeline] Batch completed for drama ${dramaId}: ${state.completedEpisodes}/${state.totalEpisodes} episodes`)
   370→   370→  }

   372→   372→  // ── Internal: Execute pipeline for a single episode ──────────
   373→   373→
   374→   374→  private async executeEpisode(
   375→   375→    episode: {
   376→   376→      id: string
   377→   377→      episodeNumber: number
   378→   378→      title: string
   379→   379→      rawContent: string | null
   380→   380→      scriptContent: string | null
   381→   381→      scriptStatus: string
   382→   382→      extractStatus: string
   383→   383→      storyboardStatus: string
   384→   384→      globalAssetsImported: boolean
   385→   385→      sourceChapterIds: string | null
   386→   386→    },
   387→   387→    dramaId: string,
   388→   388→    stepsToExecute: PipelineStep[],
   389→   389→    signal: AbortSignal,
   390→   390→    batchState: BatchState,
   391→   391→    episodeIndex: number
   392→   392→  ): Promise<void> {
   393→   393→    for (let stepIdx = 0; stepIdx < stepsToExecute.length; stepIdx++) {
   394→   394→      // Check abort
   395→   395→      if (signal.aborted) throw new Error('Batch was aborted')
   396→   396→
   397→   397→      // Check pause
   398→   398→      await this.checkPause(dramaId)
   399→   399→      if (batchState.status === 'failed') throw new Error('Batch failed')
   400→   400→
   401→   401→      const step = stepsToExecute[stepIdx]
   402→   402→      const stepLabel = STEP_LABELS[step]
   403→   403→      batchState.currentStep = stepLabel
   404→   404→      batchState.episodes[episodeIndex].currentStep = stepLabel
   405→   405→
   406→   406→      console.log(`[BatchPipeline] Episode ${episode.episodeNumber}, Step: ${stepLabel}`)
   407→   407→
   408→   408→      try {
   409→   409→        await this.executeStep(episode, dramaId, step, signal)
   410→   410→        batchState.episodes[episodeIndex].completedSteps = stepIdx + 1
   411→   411→      } catch (err) {
   412→   412→        const errMsg = err instanceof Error ? err.message : String(err)
   413→   413→        console.error(`[BatchPipeline] Step ${stepLabel} failed for episode ${episode.episodeNumber}:`, err)
   414→   414→        // Mark step as failed but continue to next step
   415→   415→        // This is a soft failure — the episode continues
   416→   416→        batchState.episodes[episodeIndex].completedSteps = stepIdx + 1
   417→   417→        // 记录步骤错误明细，供前端展示与排查（不再静默丢弃）
   418→   418→        const errors = batchState.episodes[episodeIndex].stepErrors || []
   419→   419→        errors.push(`${stepLabel}: ${errMsg.slice(0, 300)}`)
   420→   420→        batchState.episodes[episodeIndex].stepErrors = errors
   421→   421→      }
   422→   422→    }
   423→   423→
   424→   424→    batchState.episodes[episodeIndex].currentStep = null
   425→   425→  }
   426→   426→
   427→   427→  // ── Internal: Execute a single pipeline step ─────────────────
   428→   428→
   429→   429→  private async executeStep(
   430→   430→    episode: {
   431→   431→      id: string
   432→   432→      episodeNumber: number
   433→   433→      title: string
   434→   434→      rawContent: string | null
   435→   435→      scriptContent: string | null
   436→   436→      scriptStatus: string
   437→   437→      extractStatus: string
   438→   438→      storyboardStatus: string
   439→   439→      globalAssetsImported: boolean
   440→   440→      sourceChapterIds: string | null
   441→   441→    },
   442→   442→    dramaId: string,
   443→   443→    step: PipelineStep,
   444→   444→    _signal: AbortSignal
   445→   445→  ): Promise<void> {
   446→   446→    switch (step) {
   447→   447→      case 'script:raw':
   448→   448→        await this.executeRawContent(episode, dramaId)
   449→   449→        break
   450→   450→      case 'script:rewrite':
   451→   451→        await this.executeScriptRewrite(episode, dramaId)
   452→   452→        break
   453→   453→      case 'script:extract':
   454→   454→        await this.executeExtract(episode, dramaId)
   455→   455→        break
   456→   456→      case 'script:voice':
   457→   457→        await this.executeVoiceAssign(episode, dramaId)
   458→   458→        break
   459→   459→      case 'script:storyboard':
   460→   460→        await this.executeStoryboard(episode, dramaId)
   461→   461→        break
   462→   462→      case 'prod:chars':
   463→   463→        await this.executeCharacterImages(episode, dramaId)
   464→   464→        break
   465→   465→      case 'prod:scenes':
   466→   466→        await this.executeSceneImages(episode, dramaId)
   467→   467→        break
   468→   468→      case 'prod:dubbing':
   469→   469→        await this.executeDubbing(episode, dramaId)
   470→   470→        break
   471→   471→      case 'prod:shots':
   472→   472→        await this.executeShotFrames(episode, dramaId)
   473→   473→        break
   474→   474→      case 'prod:videos':
   475→   475→        await this.executeVideos(episode, dramaId)
   476→   476→        break
   477→   477→      case 'prod:compose':
   478→   478→        await this.executeCompose(episode, dramaId)
   479→   479→        break
   480→   480→      case 'export:merge':
   481→   481→        await this.executeMerge(episode, dramaId)
   482→   482→        break
   483→   483→      default:
   484→   484→        console.warn(`[BatchPipeline] Unknown step: ${step}`)

   490→   490→  private async executeRawContent(
   491→   491→    episode: { id: string; rawContent: string | null; globalAssetsImported: boolean; sourceChapterIds: string | null },
   492→   492→    dramaId: string
   493→   493→  ): Promise<void> {
   494→   494→    // Check if rawContent is already filled
   495→   495→    if (episode.rawContent?.trim()) return
   496→   496→
   497→   497→    // Step 1: Import global assets if not already done
   498→   498→    if (!episode.globalAssetsImported) {
   499→   499→      try {
   500→   500→        // Import global assets — this also fills rawContent from novel if possible
   501→   501→        const ep = await db.episode.findUnique({ where: { id: episode.id } })
   502→   502→        if (ep && !ep.globalAssetsImported) {
   503→   503→          await db.episode.update({
   504→   504→            where: { id: episode.id },
   505→   505→            data: { globalAssetsImported: true },
   506→   506→          })
   507→   507→        }
   508→   508→      } catch (err) {
   509→   509→        console.warn(`[BatchPipeline] Import global assets failed:`, err)
   510→   510→      }
   511→   511→    }
   512→   512→
   513→   513→    // Step 2: If still no rawContent, try to fill from novel chapters
   514→   514→    const ep = await db.episode.findUnique({ where: { id: episode.id } })
   515→   515→    if (ep?.rawContent?.trim()) return
   516→   516→
   517→   517→    if (ep?.sourceChapterIds) {
   518→   518→      try {
   519→   519→        let chapterIds: number[] = []
   520→   520→        try {
   521→   521→          chapterIds = JSON.parse(ep.sourceChapterIds)
   522→   522→        } catch { chapterIds = [] }
   523→   523→
   524→   524→        if (chapterIds.length > 0) {
   525→   525→          const novel = await db.novel.findUnique({ where: { dramaId } })
   526→   526→          if (novel) {
   527→   527→            let chapters: Array<{ index: number; title: string; content: string }> = []
   528→   528→            try {
   529→   529→              chapters = JSON.parse(novel.chapters)
   530→   530→            } catch { chapters = [] }
   531→   531→
   532→   532→            const selected = chapters.filter((ch) => chapterIds.includes(ch.index))
   533→   533→            if (selected.length > 0) {
   534→   534→              const rawContent = selected
   535→   535→                .map((ch) => `## ${ch.title}\n\n${ch.content}`)
   536→   536→                .join('\n\n---\n\n')
   537→   537→
   538→   538→              await db.episode.update({
   539→   539→                where: { id: episode.id },
   540→   540→                data: { rawContent },
   541→   541→              })
   542→   542→            }
   543→   543→          }
   544→   544→        }
   545→   545→      } catch (err) {
   546→   546→        console.warn(`[BatchPipeline] Fill rawContent from novel failed:`, err)
   547→   547→      }
   548→   548→    }
   549→   549→
   550→   550→    // If still no rawContent, mark as skipped (we can't proceed without it)
   551→   551→    const finalEp = await db.episode.findUnique({ where: { id: episode.id } })
   552→   552→    if (!finalEp?.rawContent?.trim()) {
   553→   553→      console.warn(`[BatchPipeline] Episode ${episode.id} has no rawContent — skipping script steps`)
   554→   554→    }
   555→   555→  }

   836→   836→   * Poll an async video generation task until completion.
   837→   837→   * Returns the persisted video URL (file-storage path), or null on timeout.
   838→   838→   */
   839→   839→  private async pollAsyncVideo(
   840→   840→    dramaId: string,
   841→   841→    taskId: string,
   842→   842→    intervalMs = 10000,
   843→   843→    maxPolls = 60
   844→   844→  ): Promise<string | null> {
   845→   845→    for (let i = 0; i < maxPolls; i++) {
   846→   846→      await new Promise((r) => setTimeout(r, intervalMs))
   847→   847→      try {
   848→   848→        const res = await this.internalApi(dramaId, '/api/ai/poll-status', {
   849→   849→          category: 'video',
   850→   850→          taskId,
   851→   851→          dramaId,
   852→   852→        })
   853→   853→        if (res.status === 'completed') {
   854→   854→          return typeof res.videoUrl === 'string' ? res.videoUrl : null
   855→   855→        }
   856→   856→        if (res.status === 'failed' || res.status === 'unsupported') {
   857→   857→          throw new Error(String(res.error || '视频生成失败'))
   858→   858→        }
   859→   859→      } catch (err) {
   860→   860→        if (i === maxPolls - 1) throw err
   861→   861→      }
   862→   862→    }
   863→   863→    return null
   864→   864→  }

   908→   908→
   909→   909→  private async checkPause(dramaId: string, signal?: AbortSignal): Promise<void> {
   910→   910→    const state = this.batches.get(dramaId)
   911→   911→    if (!state || state.status !== 'paused') return
   912→   912→
   913→   913→    // Wait until resumed
   914→   914→    return new Promise<void>((resolve) => {
   915→   915→      this.pauseResolvers.set(dramaId, resolve)
   916→   916→      // Also set a timeout to re-check periodically
   917→   917→      const interval = setInterval(() => {
   918→   918→        const currentState = this.batches.get(dramaId)
   919→   919→        if (!currentState || currentState.status !== 'paused') {
   920→   920→          clearInterval(interval)
   921→   921→          resolve()
   922→   922→        }
   923→   923→      }, 2000)
   924→   924→
   925→   925→      // Allow callers to abort the wait (e.g., when the batch is cancelled)
   926→   926→      if (signal) {
   927→   927→        signal.addEventListener('abort', () => {
   928→   928→          clearInterval(interval)
   929→   929→          this.pauseResolvers.delete(dramaId)
   930→   930→          resolve()
   931→   931→        }, { once: true })
   932→   932→      }
   933→   933→    })
   934→   934→  }
   935→   935→}
